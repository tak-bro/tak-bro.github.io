# Dangerous-command denylist for the agent command guard (deny-dangerous.sh).
# One POSIX-ERE regex (grep -E) per line. Blank lines and # comments are ignored.
# Design rule: block only IRREVERSIBLE/catastrophic commands (data loss, disk
# wipe, repo deletion, token exfil). Local-destructive-but-recoverable commands
# (rm -rf node_modules, git clean, docker prune) stay ALLOWED — over-blocking
# kills agent usefulness.
# After ANY edit: run hooks/test-guard.sh — must end "failed: 0".
# Origin: adapted from davidondrej/skills global-agent-guardrails.

# Command-start boundary class includes ` and ( so substitution shells
# (`echo $(git push origin main)`, backticks) can't smuggle a command past the
# guard. git patterns tolerate a path-qualified binary (/usr/bin/git) and any
# global option word between "git" and the subcommand (--no-pager, -P, plus the
# value of -C/-c/--git-dir/…) — same prefix as GIT_PRE in deny-dangerous.sh.
# A short force flag may sit inside a bundle (-uf, -fu) — 2026-10-11 lens security.

# 1. rm aimed at root, the whole home folder, the whole /Users tree, or a
#    system directory root (/etc, /Library, …-급 — OS 재설치급 비가역. 하위 파일
#    삭제(/etc/hosts)는 매치 안 됨. 2026-08-14 §3 완화의 갭 보완: §1의 기존
#    패턴들은 시스템 디렉터리를 안 잡았다).
#    2026-10-11 lens security #3: also the everyday spellings — `--` end of options,
#    doubled slashes, a path-qualified rm (/bin/rm), a quoted string run by
#    `sh -c`, and a directory glob (/etc/*, ~/.*). A quote counts as a command
#    boundary here, so chained commands that merely QUOTE such an rm are blocked
#    too; single grep/rg/echo calls still pass through the fast path.
(^|[;&|[:space:]`("'])([^;&|[:space:]`("']*/)?rm["']?[[:space:]]+((-[a-zA-Z]+|--[a-z-]*)[[:space:]]+)*["']?/+(["']|[[:space:]]*($|[;&|)`]))
(^|[;&|[:space:]`("'])([^;&|[:space:]`("']*/)?rm["']?[[:space:]]+((-[a-zA-Z]+|--[a-z-]*)[[:space:]]+)*["']?/+(etc|var|usr|bin|sbin|private|Library|System|Applications)(/+(\.?\*)?)?(["']|[[:space:]]*($|[;&|)`]))
(^|[;&|[:space:]`("'])([^;&|[:space:]`("']*/)?rm["']?[[:space:]]+((-[a-zA-Z]+|--[a-z-]*)[[:space:]]+)*["']?/+\*
(^|[;&|[:space:]`("'])([^;&|[:space:]`("']*/)?rm["']?[[:space:]]+((-[a-zA-Z]+|--[a-z-]*)[[:space:]]+)*["']?(~|\$HOME|\$\{HOME\})["']?(/+(\.?\*)?)?(["']|[[:space:]]*($|[;&|)`]))
(^|[;&|[:space:]`("'])([^;&|[:space:]`("']*/)?rm["']?[[:space:]]+((-[a-zA-Z]+|--[a-z-]*)[[:space:]]+)*["']?(~|\$HOME|\$\{HOME\})["']?/+(\.?\*|\.\?|\.\[)
# rg --pre <program> runs the program on every file it searches, path as the
# argument — with rm as the program, the search root is deleted file by file.
# Any other --pre program (pdftotext, a decompressor) stays allowed.
(^|[;&|[:space:]`(])([^;&|[:space:]`(]*/)?rg[[:space:]][^;&|]*--pre([[:space:]]+|=)["']?([^;&|[:space:]"']*/)?(rm|unlink)["']?([[:space:];&|)`]|$)
(^|[;&|[:space:]`("'])([^;&|[:space:]`("']*/)?rm["']?[[:space:]]+((-[a-zA-Z]+|--[a-z-]*)[[:space:]]+)*["']?/+Users(/[^/[:space:]"']+)?(/+(\.?\*)?)?(["']|[[:space:]]*($|[;&|)`]))
--no-preserve-root

# 3. removed 2026-08-14 (완화): the blanket sudo-rm line blocked every
#    admin-privilege delete including `sudo rm /tmp/x`. The §1-ENUMERATED targets
#    (root/home/Users + the 8 listed system dirs — NOT /opt, /usr/local,
#    /Volumes/<drive>, /nix) are caught by §1 regardless of a sudo prefix
#    (the rm patterns anchor on the rm word, not line start). NOTE: §1 originally
#    did NOT cover system dirs — review caught that gap same day; the system-dir
#    line above closes it. Non-catastrophic sudo deletes are the user's risk to
#    take, same as unprivileged rm.

# 5. removed 2026-08-14 (3차 완화, 배포 흐름): the internet-pipe blocks are gone
#    entirely. Unprivileged curl|sh went first (standard installer idiom —
#    rustup, bun, uv); the sudo-shell variant followed because provisioning/deploy
#    scripts (get.docker.com class) use it legitimately, and the unprivileged
#    variant being allowed already conceded the "runs arbitrary remote code on
#    this account" risk — sudo adds scope, not a new class.

# 6. removed 2026-08-14 (완화): force push to a NON-protected branch is allowed —
#    the remote keeps the old SHA (reflog/local clones), so it fails the
#    "irreversible only" bar, and rebase-then-force-push is a legitimate agent
#    workflow. Force pushes that NAME a protected branch are still caught by 6b
#    (`[^;&|]*` spans the flags there).

# 6b. 2026-08-14 4차 완화 (계약 해제): plain push/commit/edit to protected branches
#     is ALLOWED — a plain push is recoverable with a revert commit, and
#     deploy-by-push-to-main is a real workflow. The same day removed: the
#     settings.json Edit|Write branch hook, the commit policy branch, and the
#     draft-only/merge policy branches in deny-dangerous.sh (same contract set).
#     What stays blocked is HISTORY REWRITE on a protected branch: a force push
#     (--force / --force-with-lease / -f / +refspec) that NAMES main/master/develop
#     in the command — the remote branch pointer moves destructively for every
#     clone. A refspec-less `git push -f` / `push -f origin HEAD` from a protected
#     checkout PASSES (text-only match, no branch awareness — README 알려진 한계).
#     Word-bounded so "maintenance"/"main-backup" stay allowed.
#     Three lines: flag-before-ref, ref-before-flag, +refspec (ERE has no lookahead).
(^|[;&|[:space:]`(])["']?([^;&|[:space:]`(]*/)?git["']?[[:space:]]+((-[Cc]|--git-dir|--work-tree|--namespace|--exec-path|--config-env|--attr-source)([[:space:]]+|=)[^;&|[:space:]]+[[:space:]]+|-[^;&|[:space:]]*[[:space:]]+)*push[^;&|]*[[:space:]](--force(-with-lease(=[^[:space:];&|]*)?)?|-[a-zA-Z0-9]*f[a-zA-Z0-9]*)[^;&|]*[[:space:]:+/]["']?(main|master|develop)["']?([[:space:];&|:)`]|$)
(^|[;&|[:space:]`(])["']?([^;&|[:space:]`(]*/)?git["']?[[:space:]]+((-[Cc]|--git-dir|--work-tree|--namespace|--exec-path|--config-env|--attr-source)([[:space:]]+|=)[^;&|[:space:]]+[[:space:]]+|-[^;&|[:space:]]*[[:space:]]+)*push[^;&|]*[[:space:]:+/]["']?(main|master|develop)["']?[^;&|]*[[:space:]](--force(-with-lease(=[^[:space:];&|]*)?)?|-[a-zA-Z0-9]*f[a-zA-Z0-9]*)([[:space:];&|)`]|$)
(^|[;&|[:space:]`(])["']?([^;&|[:space:]`(]*/)?git["']?[[:space:]]+((-[Cc]|--git-dir|--work-tree|--namespace|--exec-path|--config-env|--attr-source)([[:space:]]+|=)[^;&|[:space:]]+[[:space:]]+|-[^;&|[:space:]]*[[:space:]]+)*push[^;&|]*[[:space:]]["']?\+[^;&|[:space:]]*:?["']?(main|master|develop)["']?([[:space:];&|:)`]|$)

# 6c. removed 2026-08-14: remote TAG deletion is now allowed, same rationale as
#     branch deletion (6b note) — a deleted tag is recoverable from its SHA, so it
#     fails the "irreversible only" bar. The refs/tags/ patterns were also partial
#     by design (bare-name `git push origin --delete v1.0.0` always passed), so
#     they added friction without a guarantee. (`gh release delete` was later
#     allowed too — 3차 완화 — so tag-deletion safety now rests on SHA recovery
#     alone, not on a release-path guard.)
#
# 6d. nameless remote mass-deletion: `git push --mirror` force-syncs the remote to
#     the local ref set (deleting every remote-only branch AND overwriting
#     protected branches without naming them — bypasses 6b by construction).
#     2026-08-14 완화: the `--prune` combination patterns (--all --prune, wildcard
#     refspec) were dropped — rare commands, complex regexes, partial coverage.
#     They also delete namelessly, but the remote-only branches they prune remain
#     recoverable from local clones; --mirror additionally OVERWRITES protected
#     branches (bypassing 6b), which is why it alone keeps the block.
(^|[;&|[:space:]`(])["']?([^;&|[:space:]`(]*/)?git["']?[[:space:]]+((-[Cc]|--git-dir|--work-tree|--namespace|--exec-path|--config-env|--attr-source)([[:space:]]+|=)[^;&|[:space:]]+[[:space:]]+|-[^;&|[:space:]]*[[:space:]]+)*push[^;&|]*[[:space:]]--mirror([[:space:];&|)`]|$)

# 9. GitHub CLI (gh): irreversible or account-level destruction + token exfil.
#    2026-08-14 완화: the blanket `gh api -X/--method DELETE` line was dropped —
#    it blocked every reversible DELETE (comments, labels, reactions). Same-day
#    review correction: `gh repo delete` does NOT cover the api route, so the two
#    lines below restore ONLY the repo-root DELETE (`…/repos/<owner>/<repo>` as
#    the path END — sub-resource DELETEs stay allowed). One line per flag/path
#    order; ERE can't express "both present, any order" in one line.
(^|[;&|[:space:]`(])gh[[:space:]]+api[[:space:]][^;&|]*(-X|--method)(=|[[:space:]]*)["']?(DELETE|delete)["']?[[:space:]][^;&|]*/?repos/[^/[:space:]"';&|]+/[^/[:space:]"';&|]+/?["']?([[:space:]]|$|[;&|)`])
(^|[;&|[:space:]`(])gh[[:space:]]+api[[:space:]][^;&|]*/?repos/[^/[:space:]"';&|]+/[^/[:space:]"';&|]+/?["']?[[:space:]][^;&|]*(-X|--method)(=|[[:space:]]*)["']?(DELETE|delete)["']?([[:space:])`]|$|[;&|])
(^|[;&|[:space:]`(])gh[[:space:]]+repo[[:space:]]+delete([[:space:])`]|$)

(^|[;&|[:space:]`(])gh[[:space:]]+(secret|ssh-key|gpg-key)[[:space:]]+delete([[:space:])`]|$)
